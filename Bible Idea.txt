    const callGeminiAPI = async (prompt, systemInstruction = "", responseSchema = null) => {
      let delay = 1000;
      for (let attempt = 0; attempt < 4; attempt++) {
        try {
          const apiKey = ""; // Canvas auto-injects this key at runtime
          const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;
         
          const payload = {
            contents: [{ parts: [{ text: prompt }] }],
          };
          if (systemInstruction) {
            payload.systemInstruction = { parts: [{ text: systemInstruction }] };
          }
          if (responseSchema) {
            payload.generationConfig = {
              responseMimeType: "application/json",
              responseSchema: responseSchema
            };
          }

          const response = await fetch(apiUrl, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(payload)
          });

          if (!response.ok) {
            if (response.status === 429) {
              console.warn(`Rate limited (429). Retrying in ${delay}ms...`);
              await new Promise(resolve => setTimeout(resolve, delay));
              delay *= 2;
              continue;
            }
            throw new Error(`Failed to generate content: Status ${response.status}`);
          }

          const result = await response.json();
          if (result.candidates && result.candidates.length > 0) {
            return result.candidates[0].content.parts[0].text.trim();
          }
          throw new Error("Invalid response format");
        } catch (error) {
          console.error(`Gemini API Error (Attempt ${attempt + 1}):`, error);
          if (attempt === 3) return null;
          await new Promise(resolve => setTimeout(resolve, delay));
          delay *= 2;
        }
      }
      return null;
    };

Example format: "Esta es una oración sofisticada [sophisticated]." Provide ONLY the final text response without quotes or explanations.`;

            if (!part) return null;
            // If the part is purely whitespace or punctuation, render normally

            }

                className="thesaurus-word"
                onMouseEnter={(e) => onWordHover(part, e.currentTarget)}

                onClick={(e) => onWordHover(part, e.currentTarget, true)}

            );

                  const cacheKey = `${code}_${selectedLevel}_${translationStyle}`;
                  const translation = message.translations?.[cacheKey];

                  return (
                    <div key={code} className="relative min-h-[1.5rem]">
                      <div className="text-xs font-bold text-gray-500 dark:text-gray-400 mb-1 flex items-center gap-1.5">
                        <span className="text-sm">{targetLang.icon}</span> <span className="uppercase tracking-wider">{targetLang.name}</span>
                      </div>
                      {translation ? (
                        <div className="text-[15px] font-medium leading-relaxed animate-fade-in whitespace-pre-wrap text-indigo-700 dark:text-indigo-300">
                          <InteractiveText
                            text={translation}
                            isThesaurusActive={isThesaurusActive}
                            onWordHover={onWordHover}
                            onWordLeave={onWordLeave}
                          />
                        </div>
                      ) : (
                        <div className="flex items-center space-x-2 text-sm text-gray-500 dark:text-gray-400">
                          <div className={`loader ${isUser ? 'loader-white' : 'border-indigo-500'}`}></div>
                          <span>Translating to {targetLang.name}...</span>
                        </div>
                      )}
                    </div>
                  );
                })}
              </div>
            )}
          </div>
        </div>
      );
    };

                {isThesaurusActive && (
                  <div className="flex justify-center mb-6">
                    <div className="bg-amber-100 dark:bg-amber-950/50 text-amber-800 dark:text-amber-300 text-xs font-semibold px-4 py-2 rounded-full flex items-center shadow-sm border border-amber-200 dark:border-amber-900/50 animate-bounce">

                        onWordLeave={handleWordLeave}

                              <span className="text-[9px] font-bold text-gray-400 dark:text-gray-500 uppercase tracking-wider mb-1 block">
                                <i className="fa-solid fa-equals mr-1"></i> Synonyms

                              <div className="flex flex-wrap gap-1.5">
                                {thesaurusPopup.data.synonyms.map((s, i) => (
                                  <span key={i} className="bg-blue-50 dark:bg-blue-950/40 text-blue-700 dark:text-blue-300 border border-blue-100 dark:border-blue-900/50 px-2 py-1 rounded-lg text-[10px] font-medium">
                                    {s}
                                  </span>
                                ))}
                              </div>
                            </div>
                          )}

                          {thesaurusPopup.data.antonyms && thesaurusPopup.data.antonyms.length > 0 && (
                            <div>
                              <span className="text-[9px] font-bold text-gray-400 dark:text-gray-500 uppercase tracking-wider mb-1 block">
                                <i className="fa-solid fa-not-equal mr-1"></i> Antonyms
                              </span>
                              <div className="flex flex-wrap gap-1.5">
                                {thesaurusPopup.data.antonyms.map((a, i) => (
                                  <span key={i} className="bg-red-50 dark:bg-red-950/40 text-red-700 dark:text-red-300 border border-red-100 dark:border-red-900/50 px-2 py-1 rounded-lg text-[10px] font-medium">
                                    {a}
                                  </span>

                                <i className="fa-solid fa-quote-left mr-1"></i> Related Adjectives

                                {thesaurusPopup.data.adjectives.map((adj, i) => (
                                  <span key={i} className="bg-emerald-50 dark:bg-emerald-950/40 text-emerald-700 dark:text-emerald-300 border border-emerald-100 dark:border-emerald-900/50 px-2 py-1 rounded-lg text-[10px] font-medium">

                                ))}

                      ) : (
                        <div className="text-red-500 text-xs py-4 text-center">
                          <i className="fa-solid fa-triangle-exclamation mr-2"></i> Could not compile thesaurus.
                        </div>
                      )}
                    </div>
                  </div>
                );
              })()}

              {/* Footer Interactive Chat Input */}
              {activeMode === 'chat' && (
                <footer className="bg-white dark:bg-gray-800 p-4 border-t border-gray-150 dark:border-gray-700 z-10 transition-colors duration-200">
                  <form onSubmit={handleSendMessage} className="flex gap-3">
                    <input
                      type="text"
                      value={inputText}
                      onChange={(e) => setInputText(e.target.value)}
                      placeholder="Ask the AI study assistant about translation structures, ancient texts, or history..."
                      className="flex-1 bg-gray-50 dark:bg-gray-900 border border-gray-200 dark:border-gray-700 rounded-2xl px-5 py-3.5 focus:outline-none focus:ring-2 focus:ring-blue-500/50 focus:border-blue-500 text-gray-800 dark:text-gray-100 transition-all text-xs sm:text-sm"
                    />
                    <button
                      type="submit"
                      disabled={isChatting || !inputText.trim()}
                      className="bg-blue-600 dark:bg-blue-700 text-white rounded-2xl px-6 py-3.5 font-semibold hover:bg-blue-700 dark:hover:bg-blue-600 transition-colors disabled:opacity-50 disabled:cursor-not-allowed shadow-md flex items-center justify-center shrink-0"
                    >
                      <i className="fa-solid fa-paper-plane text-xs"></i>
                    </button>
                  </form>
                </footer>
              )}

            </div>
          </div>
        </div>
      );
    };

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
